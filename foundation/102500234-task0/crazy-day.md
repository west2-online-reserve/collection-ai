```python
import time

def muscle_growth(days, daily_shower):
    muscle = 0
    print("🧴 增肌沐浴露效果模拟开始...")
    print("=" * 40)
    
    for day in range(1, days + 1):
        if daily_shower:
            gain = 17
            muscle += gain
        else:
            gain = 0
            
        print(f"第 {day} 天：臂围变化 +{muscle:.2f} cm（原地起飞🚀）")
        time.sleep(0.5)
    
    print("=" * 40)
    if muscle >= 17:
        print("📢 结论：一天暴涨17cm！🏆")
    else:
        print("📢 结论：增肌效果 ≈ 0，建议去健身房。")

muscle_growth(1, True)
```
